1. وصف المشروع والمعمارية والبيئة التقنية

1.1 فكرة المشروع ونظرة عامة

مشروع لغة بيان (Bayan) هو نظام حاسوبي متكامل صُمم لتقديم لغة برمجة عربية عالية المستوى، مدعومة بمترجم مستقل مبني بالكامل من الصفر (From Scratch) ومحرر تفاعلي حديث يحاكي واجهة وتجربة المستخدم لبرنامج Visual Studio Code، مع دعم كامل وأصيل لاتجاه الكتابة والواجهة من اليمين إلى اليسار (RTL).

المعمارية المستقلة المنفصلة (Decoupled Architecture):
المحرر المرئي والمترجم هما برنامجان تنفيذيان منفصلان تماماً. عند النقر على تشغيل أو ترجمة، يقوم المحرر باستدعاء المترجم المستقل (BayanCompiler.exe) عبر سطر الأوامر (CLI Inter-process Communication) وتمرير مسار الملف، ثم قراءة وتحليل النتائج وعرضها في تبويبات مخصصة.

1.2 لغات البرمجة ومنصة التشغيل

  • لغة البرمجة الأساسية: C# (الإصدار C# 10.0 الحديث).
  • بيئة ومنصة التشغيل: Microsoft .NET 6.0 LTS (الإصدار 6.0.100).

2.1 المحلل المعجمي واستخراج الرموز (Lexer.cs)

بطاقة المواصفات البرمجية: المحلل المعجمي (Lexical Analyzer Specification)

2. المدخلات (Inputs)النص البرمجي المصدري المكتوب بلغة بيان بترميز UTF-8 الممرر من المحرر أو سطر الأوامر.

7. المكتبات المستخدمةSystem, System.Collections.Generic, System.Text, ومكتبة الواجهة الأمامية للمشروع BayanCompiler.Frontend.
8. الملفات المرتبطة بالجزءToken.cs, TokenType.cs, SourceText.cs, Diagnostic.cs, DiagnosticBag.cs.
9. كيفية ارتباطه بباقي المشروعيمثل المرحلة الأولى (بوابة خط الأنابيب)؛ يستقبل النص المباشر من واجهة المحرر، ويمرر مصفوفة الرموز (Tokens) مباشرة كمدخل أساسي للمحلل النحوي التنبؤي DoctorLl1Parser.cs.
// مقتطف حقيقي من المحلل المعجمي Lexer.cs
public LexResult Lex()
{
    var tokens = new List<Token>();
    while (_position < _text.Length)
    {
        char current = Current;
        if (char.IsWhiteSpace(current)) { Advance(); continue; }
        if (current == '/' && Peek(1) == '/') { SkipLineComment(); continue; }

        if (char.IsLetter(current) || current == '_')
        {
            tokens.Add(ReadIdentifierOrKeyword());
            continue;
        }
        if (char.IsDigit(current))
        {
            tokens.Add(ReadNumber());
            continue;
        }
        if (current == '"')
        {
            tokens.Add(ReadString());
            continue;
        }
        tokens.Add(ReadOperatorOrDelimiter());
    }
    tokens.Add(new Token(TokenType.EndOfFile, "\0", new SourceSpan(_position, 0, _line, _column)));
    return new LexResult(tokens, _diagnostics.ToReadOnlyList());
}

9. كيفية ارتباطه بباقي المشروعيستهلك رموز المحلل المعجمي، ويبني شجرة الإعراب المجردة (AST) التي تمثل الهيكل المحوري الممرر للمحلل الدلالي DoctorSemanticAnalyzer.cs.
// مقتطف حقيقي من التحليل النحوي التنبؤي وبناء عقد التعبيرات DoctorLl1Parser.cs
public DoctorParseResult Parse()
{
    var parseStack = new Stack<string>();
    parseStack.Push(DoctorGrammar.RuleEndMarker);
    parseStack.Push(DoctorGrammar.StartSymbol); // "برنامج"

    while (parseStack.Count > 0)
    {
        string top = parseStack.Pop();
        Token currentToken = Current;

        if (IsTerminal(top))
        {
            if (MatchTerminal(top, currentToken))
            {
                Advance(); // استهلاك الرمز بعد التأكد من صحته
            }
            else
            {
                ReportSyntaxError(top, currentToken);
                return DoctorParseResult.Failure(_diagnostics);
            }
        }
        else // رمز غير نهائي Non-terminal: تطبيق قاعدة الإنتاج من جدول LL(1)
        {
            var production = LookupProduction(top, currentToken.Type);
            for (int i = production.Count - 1; i >= 0; i--)
            {
                if (production[i] != DoctorGrammar.Epsilon)
                    parseStack.Push(production[i]);
            }
        }
    }
    return DoctorParseResult.Success(_astBuilder.Build());
}

9. كيفية ارتباطه بباقي المشروعيستقبل شجرة AST من المحلل النحوي، ويؤكد سلامتها من أخطاء الأنواع والمجالات، ويزود مولد الشفرة الوسيطة IrGenerator.cs بالأنواع المؤكدة لإنتاج كود سليم ومحسن.
// مقتطف حقيقي من التحقق الدلالي وفحص مجالات الرؤية DoctorSemanticAnalyzer.cs
public override void VisitAssignment(AssignmentNode node)
{
    var symbol = _currentScope.Lookup(node.VariableName);
    if (symbol == null)
    {
        ReportError($"المعرف '{node.VariableName}' غير معرّف في هذا النطاق.", node.Span);
        return;
    }
    if (symbol.IsReadOnly)
    {
        ReportError($"لا يمكن تعديل الثابت '{node.VariableName}' بعد تعيين قيمته الأولية.", node.Span);
        return;
    }

    LanguageType exprType = EvaluateType(node.ValueExpression);
    if (!IsTypeCompatible(symbol.Type, exprType))
    {
        ReportError($"تعارض في الأنواع: لا يمكن إسناد قيمة من نوع '{exprType}' إلى متغير من نوع '{symbol.Type}'.", node.Span);
    }
}

// مقتطف حقيقي من توليد أوامر الشفرة الوسيطة ثلاثية العناوين IrGenerator.cs
private IrValue GenerateBinaryExpression(BinaryExpressionNode binary)
{
    IrValue leftVal = GenerateExpression(binary.Left);
    IrValue rightVal = GenerateExpression(binary.Right);
    IrValue temp = NewTemporary(binary.InferredType);

    _program.Emit(new IrInstruction(
        IrOpcode.Binary,
        temp,
        leftVal,
        rightVal,
        binary.Operator,
        null,
        binary.Span
    ));
    return temp;
}

// مقتطف حقيقي من توليد شفرة لغة التجميع MipsCodeGenerator.cs
private void EmitMipsInstruction(IrInstruction inst, StringBuilder sb)
{
    switch (inst.Opcode)
    {
        case IrOpcode.Assign:
            LoadValueToReg(inst.Left, "$t0", sb);
            sb.AppendLine($"    sw $t0, {GetMemoryLocation(inst.Result)}");
            break;
        case IrOpcode.Binary:
            LoadValueToReg(inst.Left, "$t0", sb);
            LoadValueToReg(inst.Right, "$t1", sb);
            string mipsOp = GetMipsOpcode(inst.Operator);
            sb.AppendLine($"    {mipsOp} $t2, $t0, $t1");
            sb.AppendLine($"    sw $t2, {GetMemoryLocation(inst.Result)}");
            break;
        case IrOpcode.PrintInt:
            LoadValueToReg(inst.Left, "$a0", sb);
            sb.AppendLine("    li $v0, 1");       // syscall 1: طباعة عدد صحيح
            sb.AppendLine("    syscall");
            break;
    }
}

2.6 توليد الملف التنفيذي الأصلي CIL و PE EXE (CilExecutableGenerator.cs & ilasm.exe)

بطاقة المواصفات البرمجية: توليد الملف التنفيذي (PE Executable Generator Specification)
1. وظيفة الجزء من الكودتوليد كود لغة مايكروسوفت الوسيطة (Common Intermediate Language - CIL)، وضبط ترميز UTF-8 للمدخلات والمخرجات لحل مشكلة النصوص العربية، ثم استدعاء مجمع مايكروسوفت ilasm.exe لتجميع الكود وإنتاج ملف تنفيذي أصلي output.exe يعمل مباشرة في بيئة ويندوز دون محاكٍ.
2. المدخلات (Inputs)برنامج الشفرة الوسيطة IrProgram ومسار مجلد الحفظ المختار.
3. المخرجات (Outputs)ملف الكود المصدري الوسيط output.il، والملف التنفيذي النهائي المستقل output.exe (Windows Portable Executable).
4. المتغيرات المستخدمةilasmPath (مسار أداة ilasm على نظام ويندوز)، ilText (نص شفرة CIL المولدة)، locals (خريطة مواقع المتغيرات المحلية في إطار المكدس)، startInfo (معلمات إطلاق عملية المجمع).
5. هياكل البيانات المستخدمةProcessStartInfo، قواميس المتغيرات Dictionary<string, int> لترقيم مواقع CIL Local Slots، وقواميس عنونة العلامات Dictionary<string, string>.
6. الدوال أو الإجراءات المستخدمةGenerate() (تنسيق خطوات التوليد)، EmitMainHeader() (بناء ترويسة دالة Main وضبط ترميز UTF8)، CompileIlToExe() (إطلاق مجمع ilasm كعملية فرعية)، FindIlasmPath() (البحث التلقائي عن مسار مجمع مايكروسوفت).
7. المكتبات المستخدمةSystem, System.Diagnostics, System.IO, System.Text.
8. الملفات المرتبطة بالجزءCilExecutableGenerator.cs, AssemblyGenerator.cs, WindowsExecutableBuilder.cs.
9. كيفية ارتباطه بباقي المشروعيمثل محطة الإنتاج الختامية للمترجم؛ يُنتج الملف التنفيذي الحقيقي الذي يقوم المحرر باستدعائه وتشغيله تفاعلياً داخل الطرفية المدمجة.
// مقتطف حقيقي من ضبط ترميز UTF-8 وتوليد الملف التنفيذي عبر ilasm AssemblyGenerator.cs
private void EmitMainHeader(StringBuilder builder, IrProgram program)
{
    builder.AppendLine("    .method public static void Main() cil managed");
    builder.AppendLine("    {");
    builder.AppendLine("        .entrypoint");
    builder.AppendLine("        .maxstack 32");
    builder.AppendLine();
    // فرض ترميز UTF-8 في دالة المدخل لضمان طباعة وقراءة النصوص العربية
    builder.AppendLine("        call class [mscorlib]System.Text.Encoding [mscorlib]System.Text.Encoding::get_UTF8()");
    builder.AppendLine("        call void [mscorlib]System.Console::set_OutputEncoding(class [mscorlib]System.Text.Encoding)");
    builder.AppendLine("        call class [mscorlib]System.Text.Encoding [mscorlib]System.Text.Encoding::get_UTF8()");
    builder.AppendLine("        call void [mscorlib]System.Console::set_InputEncoding(class [mscorlib]System.Text.Encoding)");
}

2.7 محرر لغة بيان وبيئة التطوير والطرفية التفاعلية (CompilerJourneyForm.cs & BayanTerminalControl.cs)

بطاقة المواصفات البرمجية: محرر اللغة والطرفية (Language Editor & Terminal Specification)

9. كيفية ارتباطه بباقي المشروعيمثل الواجهة المرئية الجامعة للنظام بأكمله؛ يحقق معيار الفصل التام (Decoupling) باستدعاء المترجم المستقل كعملية خارجية واستقبال مخرجاته، ثم تشغيل الملف التنفيذي وإدارته داخل الطرفية.
// مقتطف حقيقي من استدعاء المترجم المستقل وعرض نتائج المراحل CompilerJourneyForm.cs
private async void CompileSource()
{
    SetBusy(true, "يجري تشغيل مترجم بيان المستقل والتحقق من القواعد...");
    try
    {
        // استدعاء المترجم المستقل عبر IPC
        CliCompilationRun run = await compilerClient.CompileAsync(sourceEditor.SourceCode);

        // توزيع مخرجات المراحل على تبويبات الفاحص
        txtTokens.Text = run.GetArtifact("tokens.txt");
        txtParseTree.Text = run.GetArtifact("parse-tree.txt");
        txtSymbolTable.Text = run.GetArtifact("symbol-table.txt");
        txtTac.Text = run.GetArtifact("three-address-code.txt");
        txtAssembly.Text = run.GetArtifact("output.asm");

        lastExecutablePath = run.Manifest?.WindowsExecutablePath ?? string.Empty;
        btnQuickRun.Enabled = run.Manifest?.Success == true && File.Exists(lastExecutablePath);
    }
    finally { SetBusy(false); }
}
